# ---------------- TICKET EMBED ----------------

def build_ticket_embed(user, data):
    embed = discord.Embed(
        title="🎨 New Graphic Design Ticket",
        color=discord.Color.gold()
    )

    embed.add_field(
        name="👤 User",
        value=user.mention,
        inline=False
    )

    embed.add_field(
        name="🎨 Design Type",
        value=data["design"],
        inline=False
    )

    embed.add_field(
        name="📝 Description",
        value=data["description"],
        inline=False
    )

    embed.add_field(
        name="✨ Style",
        value=data["style"],
        inline=False
    )

    embed.add_field(
        name="🌈 Colors",
        value=data["colors"],
        inline=False
    )

    embed.add_field(
        name="📐 Dimensions",
        value=data["size"],
        inline=False
    )

    embed.add_field(
        name="⏰ Deadline",
        value=data["deadline"],
        inline=False
    )

    embed.set_footer(text="A designer will assist you shortly.")

    return embed
